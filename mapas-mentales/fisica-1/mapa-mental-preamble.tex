% ---------- Estilo base de los conceptos (nodos) ----------
% Nivel raíz: el examen/tema central.
% Nivel 1: familias de conceptos (ramas, coloreadas).
% Nivel 2: conceptos individuales (hojas, color heredado de su rama).
%
% grow cyclic es obligatorio: sin él las ramas no se distribuyen en
% círculo, crecen todas hacia abajo y se apilan una sobre otra.
% Va en ambos niveles para que las hojas también se abran en abanico
% alrededor de su rama.
%
% text width fija el ancho de párrafo dentro del círculo; sin él, una
% etiqueta larga se desborda del nodo o lo infla de forma desproporcionada.
% minimum size va un poco por encima de text width para que quede aire.
\tikzstyle{every node}=[font=\sffamily]
\tikzset{
  root concept/.style={
    concept color=textgray,
    font=\sffamily\bfseries\Large,
    text=white,
    minimum size=3.6cm,
    text width=3.0cm,
  },
  level 1 concept/.append style={
    grow cyclic,
    font=\sffamily\bfseries\small,
    text=white,
    level distance=5.4cm,
    minimum size=2.7cm,
    text width=2.3cm,
  },
  level 2 concept/.append style={
    grow cyclic,
    font=\sffamily\scriptsize,
    text=white,
    % La cuerda entre dos hojas vecinas es 2*d*sin(a/2); con d=3.6cm y a=50
    % da unos 3cm, por encima del diámetro de la hoja (2.1cm). Con menos
    % ángulo o menos distancia, las hojas se pisan entre sí.
    sibling angle=50,
    level distance=3.6cm,
    minimum size=2.1cm,
    text width=1.8cm,
  },
}

% ---------- Número de ramas ----------
% El ángulo entre ramas de nivel 1 tiene que ser 360/n, y n (cantidad de
% familias) solo lo sabe el cuerpo del documento. generador-mapa-mental
% DEBE llamar \setRamas{<n>} antes del tikzpicture. El default de 4 es
% solo para que el preámbulo compile solo; con otra cantidad de ramas, sin
% \setRamas el mapa queda volcado hacia un lado.
\newcommand{\setRamas}[1]{%
  \tikzset{level 1 concept/.append style={sibling angle=360/#1}}%
}
\setRamas{4}
